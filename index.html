let realtimeEnabled = false; let channel = null; function enableRealtime() { if (realtimeEnabled) { console.log("Realtime already enabled"); return; } channel = supabase.channel("live-feed"); channel .on( "postgres_changes", { event: "*", schema: "public", table: "feed" }, (payload) => { console.log("🔥 feed update:", payload); load(); } ) .on( "postgres_changes", { event: "*", schema: "public", table: "places" }, (payload) => { console.log("📍 places update:", payload); load(); } ) .on( "postgres_changes", { event: "*", schema: "public", table: "geofences" }, (payload) => { console.log("🟠 geofence update:", payload); load(); } ) .subscribe(); realtimeEnabled = true; console.log("✅ Realtime enabled"); }
